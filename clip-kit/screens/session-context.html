f/ops/retry_payment.ts
export async function main(stripe: RT.Stripe, payment_id: string) {
  const p = await getPayment(stripe, payment_id)
  if (p.attempts >= 3) return { status: "gave_up" }
  const r = await confirmPayment(stripe, payment_id)
  return { payment: payment_id, attempt: p.attempts + 1, status: r.status }
}
